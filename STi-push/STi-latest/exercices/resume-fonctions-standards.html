🔤 Fonction alpha(ch) JS

Vérifie si une chaîne est composée uniquement de lettres alphabétiques et d'espaces. 

function alpha(ch){
  const alphabet = " AZERTYUIOPQSDFGJKLMWXCVBN";
  let nb = 0;
  for(let i = 0; i < ch.length; i++) {
    if (alphabet.indexOf(ch.charAt(i)) >= 0) nb++;
  }
  return (ch.length == nb);
}

📝 Fonction verifnom() JS

Vérifie que le champ nom/prénom ne contient que des lettres/espaces et un maximum de 5 mots. 

function verifnom(){
  let np = document.getElementById("np").value;
  let ok = true;
  np = np.trim();
  document.getElementById("np").value = np;
  document.getElementById("npa").innerHTML = "";
  if ((alpha(np.toUpperCase())) || (np.length == 0)) {
    document.getElementById("np").select();
    document.getElementById("npa").style.color = "red";
    document.getElementById("npa").innerHTML = "Attention";
    ok = false;
  }
  let tableauomt = np.split(" ");
  let tailletableaumont = tableauomt.length;
  if (tailletableaumont > 5) {
    document.getElementById("npa").style.color = "red";
    document.getElementById("npa").innerHTML = "Attention le nombre de mots est plus que 5";
    ok = false;
  }
  if (ok) {
    document.getElementById("np").style.backgroundColor = "green";
    document.getElementById("np").style.color = "white";
    document.getElementById("np").disabled = true;
  }
  return ok;
}

🔢 Fonction numerique(ch) JS

Vérifie si une chaîne est composée uniquement de chiffres.

function numerique(ch){
  let convers = Number.parseInt(ch, 10);
  return (convers.toString() == ch);
}
